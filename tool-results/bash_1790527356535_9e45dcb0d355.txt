   355→/** رسم وحدة سكنية كاملة بالمرايا الأفقية/الرأسية */
   356→function ApartmentUnit({ fx, fy }: { fx: boolean; fy: boolean }) {
   357→  const tx = (x: number, w: number) => (fx ? 20 - x - w : x);
   358→  const ty = (y: number, h: number) => (fy ? 15 - y - h : y);
   359→  const px = (x: number) => (fx ? 20 - x : x);
   360→  const py = (y: number) => (fy ? 15 - y : y);
   361→  const rot = (r: number) => (fx && fy ? 180 + r : fx ? 180 - r : fy ? -r : r);
   362→
   363→  return (
   364→    <g>
   365→      {/* الشرفة الركنية الكابولية — خارج البصمة، درابزين كامل */}
   366→      <g>
   367→        <rect
   368→          x={tx(0, BALCONY.northW)}
   369→          y={ty(-BALCONY.northD, BALCONY.northD)}
   370→          width={BALCONY.northW}
   371→          height={BALCONY.northD}
   372→          fill="#fef3c7"
   373→          opacity={0.9}
   374→          stroke="#d97706"
   375→          strokeWidth={0.05}
   376→          strokeDasharray="0.18 0.12"
   377→        />
   378→        <rect
   379→          x={tx(-BALCONY.sideW, BALCONY.sideW)}
   380→          y={ty(0, BALCONY.sideD)}
   381→          width={BALCONY.sideW}
   382→          height={BALCONY.sideD}
   383→          fill="#fef3c7"
   384→          opacity={0.9}
   385→          stroke="#d97706"
   386→          strokeWidth={0.05}
   387→          strokeDasharray="0.18 0.12"
   388→        />
   389→        {/* الدرابزين على الحواف الخارجية */}
   390→        <g stroke="#b45309" strokeWidth={0.09}>
   391→          <line x1={px(0)} y1={py(-BALCONY.northD)} x2={px(BALCONY.northW)} y2={py(-BALCONY.northD)} />
   392→          <line x1={px(BALCONY.northW)} y1={py(-BALCONY.northD)} x2={px(BALCONY.northW)} y2={py(0)} />
   393→          <line x1={px(-BALCONY.sideW)} y1={py(0)} x2={px(-BALCONY.sideW)} y2={py(BALCONY.sideD)} />
   394→          <line x1={px(-BALCONY.sideW)} y1={py(BALCONY.sideD)} x2={px(0)} y2={py(BALCONY.sideD)} />
   395→        </g>
   396→        <PlanLabel x={px(1.6)} y={py(-0.6)} size={0.24} weight={700} fill="#b45309">
   397→          شرفة ركنية
   398→        </PlanLabel>
   399→      </g>
   400→      {/* الفراغات */}
   401→      {APARTMENT_LAYOUT.map((room) =>
   402→        room.parts.map((p, i) => (
   403→          <rect
   404→            key={`${room.name}-${i}`}
   405→            x={tx(p.x, p.w)}
   406→            y={ty(p.y, p.h)}
   407→            width={p.w}
   408→            height={p.h}
   409→            fill={room.color}
   410→            stroke="#a8a29e"
   411→            strokeWidth={0.045}
   412→          />
   413→        )),
   414→      )}
   415→      {/* العناوين */}
   416→      {APARTMENT_LAYOUT.map((room) => {
   417→        const lb = UNIT_LABELS[room.name];
   418→        if (!lb) return null;
   419→        return (
   420→          <PlanLabel key={room.name} x={px(lb.lx)} y={py(lb.ly)} size={lb.ls} weight={700} fill="#57534e">
   421→            {lb.short}
   422→          </PlanLabel>
   423→        );
   424→      })}
   425→      {/* النوافذ */}
   426→      {UNIT_WINDOWS.map((wn, i) => (
   427→        <WindowSeg key={`w-${i}`} x={tx(wn.x, wn.w)} y={ty(wn.y, wn.h)} w={wn.w} h={wn.h} />
   428→      ))}
   429→      {/* الأبواب الداخلية */}
   430→      {UNIT_DOORS.map((d, i) => (
   431→        <Door key={`d-${i}`} x={px(d.x)} y={py(d.y)} r={d.r} rot={rot(d.rot)} />
   432→      ))}
   433→    </g>
   434→  );
   435→}
